\begin{frame}{Motivation}

\vskip2pt
\begin{tcolorbox}[colback=TUMBlue!8, colframe=TUMBlue, boxrule=0.8pt, arc=3pt]
\centering
{\large Recover \textbf{camera poses} ($R$, $t$) and \textbf{calibration} ($\mathbf{K}$) from \textbf{unlabeled casual video}}
\end{tcolorbox}

\vskip10pt
\textbf{Why is this hard?}
\begin{itemize}
  \item Casual video: dynamic objects, unknown cameras, no 3D labels
  \item Classical SfM/SLAM fails on dynamic scenes
  \item Supervised methods need expensive ground truth that doesn't scale
\end{itemize}

\vskip10pt
\textbf{What exists today:}
\begin{itemize}
  \item Strong pretrained models for individual sub-tasks: depth, optical flow, calibration, pose
  \item But they operate \alert{in isolation} --- no joint reasoning, no temporal consistency
\end{itemize}

\end{frame}

\begin{frame}{Contributions}

\vfill

\textbf{1. Multi-Frame Calibration Transformer (MCT)}
\begin{itemize}
  \item Cross-frame attention on calibration backbone features (4 scales)
  \item Aggregates $N$ frames $\to$ 1 consistent calibration
  \item Architecture-agnostic: inserts between any backbone and decoder
  \item Only ${\sim}25$M trainable params (7.5\% of total)
\end{itemize}

\vskip16pt
\textbf{2. Demonstrated on AnyCam + AnyCalib}
\begin{itemize}
  \item Fuses a self-supervised pose estimator with a calibration network
  \item Improves \emph{both} pose accuracy and calibration accuracy
  \item Fully self-supervised --- no ground truth labels
\end{itemize}

\vfill

\end{frame}

% SLIDE 9: Training Losses
% ============================================================================
\begin{frame}{Training Losses}

\vskip8pt
\textbf{Flow Reprojection} (self-supervised, main signal):
\begin{equation*}
  \mathbf{x}' = \pi\!\left(\mathbf{K}\left(R\,\mathbf{K}^{-1}\!\begin{pmatrix} x \\ y \\ 1\end{pmatrix}\!d + \mathbf{t}\right)\right), \quad \mathcal{L}_\text{flow} = \|\mathbf{x}' - \mathbf{x} - \mathbf{w}_\text{obs}\|
\end{equation*}

\vskip8pt
\textbf{Calibration Anchor} --- why $\mathbf{K}$ needs constraints:
\begin{equation*}
  R \approx I,\; \mathbf{t} \approx 0 \;\;\implies\;\; \mathbf{K}^{-1}(d\,\mathbf{K}\mathbf{x} + \mathbf{t}) \approx \mathbf{x} \quad \text{\small (any $\mathbf{K}$ works)}
\end{equation*}
\vskip-2pt
\begin{equation*}
  \mathcal{L}_\text{anchor} = \|\mathbf{K}_\text{pred} - \mathbf{K}_\text{AnyCalib}\|
\end{equation*}

\end{frame}

\begin{frame}{Conclusion \& Future Work}

\vskip10pt
\textbf{Summary:}
\begin{itemize}
  \item Proposed a framework for fusing pretrained calibration \& pose models
  \item MCT: multi-frame calibration consistency via feature-level aggregation
  \item Demonstrated on AnyCam + AnyCalib: \textbf{up to 40\% better translation, up to 32\% better calibration}
\end{itemize}

\vskip8pt
\textbf{Future work:}
\begin{itemize}
  \item Longer sequences --- our model trained at $N=4$ frames only
  \item End-to-end training with all components unfrozen, on larger and more diverse data
\end{itemize}
